### 2.2  Java

- [示例](https://open.feishu.cn/document/server-side-sdk/java-sdk-guide/invoke-server-api)
- [更多示例](https://github.com/larksuite/oapi-sdk-java/blob/v2_main/sample/src/main/java/com/lark/oapi/sample/api/ImSample.java)

```java
package Object.AliyunTest;

import com.lark.oapi.Client;
import com.lark.oapi.core.exception.ClientTimeoutException;
import com.lark.oapi.core.exception.ServerTimeoutException;
import com.lark.oapi.core.utils.Jsons;
import com.lark.oapi.service.im.v1.enums.MsgTypeEnum;
import com.lark.oapi.service.im.v1.enums.ReceiveIdTypeEnum;
import com.lark.oapi.service.im.v1.model.*;
import com.lark.oapi.service.im.v1.model.ext.*;

public class CreateMessageSample {
    public static void main(String[] args) {
        Client client = Client.newBuilder("appId", "appSecret").build(); // 默认配置为自建应用

        // 创建请求对象
        CreateMessageReq req = CreateMessageReq.newBuilder()
                .receiveIdType(ReceiveIdTypeEnum.OPEN_ID.getValue())
                .createMessageReqBody(CreateMessageReqBody.newBuilder()
                        .receiveId("ou_…")                                        // 指定目标用户的 open_id
                        .msgType(MsgTypeEnum.MSG_TYPE_TEXT.getValue())
                        .content(MessageText.newBuilder()                           // 消息内容构建
                                .atUser("ou_…", "tom")                // @ 某个用户，参数为 open_id 和显示名
                                .text(" test content")                              // 追加普通文本
                                .newLine()                                          // 换行
                                .textLine("jiaduo")                                 // 相当于 newLine() + text("jiaduo")
                                .atAll()                                            // @ 所有人
                                .text("注意")                                        // 追加普通文本
                                .build())
                        .build())
                .build();

        // 尝试发起请求
        try{
            // 发起请求
            CreateMessageResp resp = client.im().message().create(req);
            // 判断业务状态：飞书API统一返回 `code` 字段
            // - 0 表示成功
            // - 非 0 表示业务异常
            if(resp.getCode() != 0){
                System.out.println(String.format("code:%d, msg:%s, err:%s", resp.getCode(), resp.getMsg(), Jsons.DEFAULT.toJson(resp.getError())));
                return;
            }

            // 业务处理结果：resp.getData() 包含服务端返回的消息对象（如消息 ID、创建时间等）。
            System.out.println(Jsons.DEFAULT.toJson(resp.getData()));
            // 原生http信息：resp.getRawResponse() 包含状态码、响应头等原始信息。
            System.out.println(Jsons.DEFAULT.toJson(resp.getRawResponse()));
            // 返回请求ID：resp.getRequestId() 是飞书生成的请求追踪 ID，排查问题时很有用。
            System.out.println(resp.getRequestId());
        } catch (ServerTimeoutException e) {
            // 服务端超时异常处理
            System.out.println(e.getMessage());
            return;
        } catch (ClientTimeoutException e) {
            // 客户端超时异常处理
            System.out.println(e.getMessage());
            return;
        } catch (IllegalArgumentException e) {
            // 参数非法异常处理
            System.out.println(e.getMessage());
            return;
        } catch (Exception e) {
            // 其他异常处理
            System.out.println(e.getMessage());
            return;
        }
    }
}
```
