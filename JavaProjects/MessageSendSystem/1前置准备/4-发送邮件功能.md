#### 2.2.2  Java

```java
package Object.AliyunTest;

import jakarta.mail.*;
import jakarta.mail.internet.*;
import java.util.Properties;

public class EmailSender {

    public static void main(String[] args) {
        /* 1. 配置 SMTP 服务器信息 */
        String smtpHost = "smtp.example.com";               // SMTP 服务器地址
        String port = "465";                                // SMTP 端口，通常为 465 或 587
        String userName = "Your_Email_Address@example.com"; // 发件人邮箱地址
        String password = "Your_Password";                  // 发件人邮箱密码或授权码

        /*  2. 收件人信息 */
        String recipient = "recipient_Email_Address&example.com";

        /* 3. 邮件内容 */
        String subject = "Message_Subject";     // 邮件主题
        String body = "Message_Body";           // 邮件正文

        /* 4. 设置连接属性 */
        Properties props = new Properties();
        props.put("mail.smtp.host", smtpHost);                      // SMTP 服务器地址
        props.put("mail.smtp.port", port);                          // SMTP 端口号
        props.put("mail.smtp.auth", "true");                        // 启用身份验证
        props.put("mail.smtp.starttls.enable", "true");             // 启用 TLS 身份加密
        // props.put("mail.smtp.ssl.trust", "smtp.example.com");    // 某些环境需要信任证书

        /* 5. 创建会话对象 */
        Session session = Session.getInstance(props, new Authenticator() {
            @Override
            protected PasswordAuthentication getPasswordAuthentication() {
                return new PasswordAuthentication(userName, password);
            }
        });

        try{
            /* 6. 创建邮件对象 */
            Message message = new MimeMessage(session);
            message.setFrom(new InternetAddress(userName));                                     // 设置发件人
            message.setRecipients(Message.RecipientType.TO, InternetAddress.parse(recipient));  // 设置收件人
            message.setSubject(subject);                                                        // 设置邮件主题
            message.setText(body);                                                             // 设置邮件正文

            /* 7. 发送邮件 */
            Transport.send(message);
            System.out.println("Sent message successfully….");
            
        }  catch (MessagingException e) {
            System.out.println("Message sending failed….");
            throw new RuntimeException(e);
        }
    }
}
```

```java
package Object.AliyunTest;

import com.aliyuncs.DefaultAcsClient;
import com.aliyuncs.IAcsClient;
import com.aliyuncs.dm.model.v20151123.SingleSendMailRequest;
import com.aliyuncs.dm.model.v20151123.SingleSendMailResponse;
import com.aliyuncs.exceptions.ClientException;
import com.aliyuncs.exceptions.ServerException;
import com.aliyuncs.profile.DefaultProfile;
import com.aliyuncs.profile.IClientProfile;
public class DirectMailExample {
    public static void main(String[] args) {
        // AccessKey
        String accessKeyId = "YourAccessKeyId";
        String accessKeySecret = "YourAccessKeySecret";

        // 地区 实际的邮件推送地域
        String regionId = "cn-hangzhou";

        // 配置阿里云客户端
        IClientProfile profile = DefaultProfile.getProfile(regionId, accessKeyId, accessKeySecret);
        IAcsClient client = new DefaultAcsClient(profile);

        // 创建发送邮件的请求
        SingleSendMailRequest request = new SingleSendMailRequest();

        try{
            // 设置请求信息
            request.setAccountName("notice@yourdomain.com");                    // 设置发送地址（必须是在阿里云控制台“发信域名”和“发信地址”中已配置并验证过的真实邮箱地址，否则调用会报错。）
            request.setFromAlias("Your Alias");                                 // 发件人别名，显示名称，可选
            request.setToAddress("recipient@example.com");                      // 收件人地址，支持多个，用逗号分隔
            request.setSubject("This is the subject");                          // 设置邮件主题
            request.setHtmlBody("<html><body>This is the body</body></html>");  // 邮件正文（支持HTML）
            request.setTagName("html");                                         // 发信类型 0：文本邮件 1: HTML邮件
            request.setReplyToAddress(true);                                    // 是否允许恢复到发件人地址

            // 发送请求，并获取响应
            /* https://www.alibabacloud.com/help/en/direct-mail/api-dm-2015-11-23-singlesendmail?spm=a2c63.p38356.0.i9 */
            SingleSendMailResponse response = client.getAcsResponse(request);
            System.out.println("Email sent! Request ID: " + response.getRequestId());

        }catch (ClientException e) {
            System.out.println("Error sending email. " + e.getMessage());
            throw new RuntimeException(e);
        }

    }
}
```
