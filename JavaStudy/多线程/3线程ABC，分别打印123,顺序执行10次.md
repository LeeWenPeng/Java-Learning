public class SequentialPrinting {
    private int count = 0; // 打印计数器
    private final Object lock = new Object();

    public static void main(String[] args) {
        SequentialPrinting printing = new SequentialPrinting();

        Thread t0 = new Thread(() -> printing.printNumber(1), "A");
        Thread t1 = new Thread(() -> printing.printNumber(2), "B");
        Thread t2 = new Thread(() -> printing.printNumber(3), "C");

        t0.start();
        t1.start();
        t2.start();
    }
    private void printNumber(int flag){
        for(int i = 0; i < 10; i++){
            synchronized(lock){
                while(count % 3 != flag-1){
                    try {
                        lock.wait();
                    } catch (InterruptedException e) {
                        e.printStackTrace();
                    }
                }
                if(count < 30 && count % 3 == flag-1){
                    System.out.println("Thread " + Thread.currentThread().getName()+ " print: " + flag);
                    count++;
                    lock.notifyAll();
                }
            }
        }
    }
}
