## 1  题目

- 两个线程交替打印
- 线程一打印小写字母
- 线程二打印数字
- 目标：a1b2…z26

## 2  代码

```java
package cn.msgcenter.test;

/**
 * @description: TODO
 * @author: Lee
 * @version: 1.0
 * @date: 2026-08-30 00:35
 */
public class AlternatePrinting2 {
    private static final Object lock = new Object(); // 锁
    private static int count = 1;
    private static boolean printLetter = true;

    public static void main(String[] args) {
        Thread printLettersThread = new Thread(() -> print(true));
        Thread printNumbersThread = new Thread(() -> print(false));

        printLettersThread.start();
        printNumbersThread.start();
    }

    private static void print(boolean isPrintLetter) {
        while(count <= 26){
            synchronized (lock){
                while(printLetter != isPrintLetter){
                    try {
                        lock.wait();
                    } catch (InterruptedException e) {
                        e.printStackTrace();
                    }
                }

                if(count > 26) break;

                if(isPrintLetter){
                    System.out.println((char)('a' + count - 1));
                }else{
                    System.out.println(count);
                    count++;
                }

                printLetter = !printLetter;

                lock.notifyAll();
            }
        }
    }
}
```

```java
package cn.msgcenter.test;

/**
 * @description: TODO
 * @author: Lee
 * @version: 1.0
 * @date: 2026-08-30 00:35
 */
public class AlternatePrinting2 {
    private static final Object lock = new Object(); // 锁
    private static int count = 1;
    private static boolean printLetter = true;

    public static void main(String[] args) {
        Thread printLettersThread = new Thread(() -> {
            while( count <= 26 ){
                synchronized (lock) {
                    while(!printLetter){
                        try {
                            lock.wait();
                        } catch (InterruptedException e) {
                            throw new RuntimeException(e);
                        }
                    }

                    if( count > 26 ) break;

                    System.out.println((char)('a' + count - 1));
                    printLetter = !printLetter;

                    lock.notifyAll();
                }
            }
        });
        // 打印字母
        Thread printNumbersThread = new Thread(() -> {
            while( count <= 26 ){
                synchronized (lock) {
                    while(printLetter){
                        try {
                            lock.wait();
                        }catch (InterruptedException e) {
                            e.printStackTrace();
                        }
                    }

                    if( count > 26 ) break;
                    System.out.println(count);
                    printLetter = !printLetter;
                    count++;

                    lock.notifyAll();
                }
            }
        });

        printLettersThread.start();
        printNumbersThread.start();
    }

}
```
