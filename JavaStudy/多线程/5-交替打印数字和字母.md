## 1 题目

- 两个线程交替打印
- 线程一：打印数字，每次打印两个
- 线程二：打印字母
- 目标：`12A34B56C`

## 2 思路一：`wait` 和 `notifyAll`

```java
/**
 * @description: 使用 wait 和 notifyAll
 * @author: Lee
 * @version: 1.0
 * @date: 2026-08-29 22:25
 */
public class CrossPrint {
    private static final Object lock = new Object();
    private static boolean printNumber = true;

    public static void main(String[] args) {
        // 线程1: 打印数字
        Thread printNumberThread = new Thread(() -> {
            for(int i = 0; i < 52; i+=2) {
                synchronized (lock) {
                    while(!printNumber) {
                        try {
                            lock.wait();
                        } catch (InterruptedException e) {
                            e.printStackTrace();
                        }
                    }
                    System.out.println(i);
                    System.out.println(i+1);
                    printNumber = false;
                    lock.notifyAll();
                }
            }
        });

        // 线程2: 打印字母
        Thread printLetterThread = new Thread(() -> {
            for(char c = 'A'; c <= 'Z'; c++){
                synchronized (lock) {
                    while(printNumber){
                        try {
                            lock.wait();
                        } catch (InterruptedException e) {
                            e.printStackTrace();
                        }
                    }
                    System.out.println(c);
                    printNumber = true;
                    lock.notifyAll();
                }
            }
        });

        // 线程启动
        printNumberThread.start();
        printLetterThread.start();
    }
}
```

## 3 `ReentrantLock` 和 `Condition`

```java
package cn.msgcenter.test;

import java.util.concurrent.locks.Condition;
import java.util.concurrent.locks.Lock;
import java.util.concurrent.locks.ReentrantLock;

/**
 * @description:
 *  使用 ReentrantLock 和 Condition 实现线程之间的协调
 * @author: Lee
 * @version: 1.0
 * @date: 2026-08-29 22:37
 */
public class CrossPrintWithLock {
    private static final Lock lock = new ReentrantLock();                       // 互斥锁
    private static final Condition printNumbersCondition = lock.newCondition(); // 条件变量
    private static final Condition printLettersCondition = lock.newCondition(); // 条件变量
    private static boolean printNumber = true;                                  // 状态标志

    public static void main(String[] args) {

        // 线程1 打印数字
        Thread printNumbersThread = new Thread(() -> {
            for(int i = 1; i <= 52; i += 2){
                lock.lock();
                try{
                    while(!printNumber){                // 如果不是打印数字的回合，则等待
                        printNumbersCondition.await();
                    }
                    System.out.println(i);              // 打印数字
                    System.out.println(i+1);
                    printNumber = false;                // 更新打印状态标志
                    printLettersCondition.signal();     // 唤醒字母打印线程
                }catch (InterruptedException e) {
                    Thread.currentThread().interrupt(); // 恢复中断状态，让上层处理
                }finally {
                    lock.unlock();
                }
            }
        });

        // 线程2 打印字母
        Thread printLettersThread = new Thread(() -> {
            for(char c = 'A'; c <= 'Z'; c++){
                lock.lock();
                try{
                    while(printNumber){                 // 如果不是打印字母的回合，则等待
                        printLettersCondition.await();
                    }
                    System.out.println(c);              // 打印字母
                    printNumber = true;                 // 更新打印状态标志
                    printNumbersCondition.signal();     // 唤醒数字打印线程
                }catch (InterruptedException e) {
                    Thread.currentThread().interrupt(); // 恢复中断状态
                }finally {
                    lock.unlock();
                }
            }
        });
        printNumbersThread.start();
        printLettersThread.start();
    }
}
```
