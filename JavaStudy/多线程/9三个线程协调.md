## 3  join + CountDownLatch

```java
package cn.msgcenter.test;

import java.util.concurrent.CountDownLatch;

/**
 * @description: TODO
 * @author: Lee
 * @version: 1.0
 * @date: 2026-08-30 14:09
 */
public class ThreadExecutionOrderDemo {
    public static void main(String[] args) {
        // 创建两个 CountDownLatch 实例
        CountDownLatch t1ToT2Latch = new CountDownLatch(1); // 用于 T1 -> T2 同步
        CountDownLatch t2ToT3Latch = new CountDownLatch(1); // 用于 T2 -> T3 同步

        // t1
        Thread t1 = new Thread(() -> {
            System.out.println("T1 开始执行");
            try{
                Thread.sleep(1000);
            }catch (InterruptedException e){
                e.printStackTrace();
            }
            System.out.println("T1 执行完毕");
            t1ToT2Latch.countDown();            // T1 完成后，计数减 1,允许 T2 开始
        });

        Thread t2 = new Thread(() -> {
            try {
                t1ToT2Latch.await();            // 等待 T1 完成
            }catch (InterruptedException e){
                e.printStackTrace();
            }
            System.out.println("T2 开始执行");
            try{
                Thread.sleep(1000);         // T2 todo
            }catch (InterruptedException e){
                e.printStackTrace();
            }
            System.out.println("T2 执行完成");
            t2ToT3Latch.countDown();
        });

        Thread t3 = new Thread(() -> {
            try{
                t2ToT3Latch.await();
            }catch (InterruptedException e){
                e.printStackTrace();
            }

            System.out.println("T3 开始执行");
            try{
                Thread.sleep(1000);
            }catch (InterruptedException e){
                e.printStackTrace();
            }
            System.out.println("T3 执行完成");
        });

        t1.start();
        t2.start();
        t3.start();

        try{
            t1.join();
            t2.join();
            t3.join();
        }catch (InterruptedException e){
            e.printStackTrace();
        }

        System.out.println("所有任务执行完成");
    }
}
```
