- 总共有 500 张票
- 有 4 个购票窗口：1、2、3、4
- 模拟购票流程，打印购票结果
- 比如：从窗口 1 购票 1 张，剩余 499 张。

```java
package cn.msgcenter.test;

/**
 * @description: TODO
 * @author: Lee
 * @version: 1.0
 * @date: 2026-08-30 18:49
 */
public class TicketSystemDemo {
    // 总票数
    private static final int TOTAL_TICKETS = 500;
    // 剩余票数
    private static int remainingTickets = TOTAL_TICKETS;
    // 锁对象
    private static Object lock = new Object();

    public static void main(String[] args) {
        // 创建四个购票窗口的线程
        for(int i = 1; i <= 4; i++){
            new Thread(new TickWindow(i)).start();
        }
    }
    static class TickWindow implements Runnable{
        private int id;
        TickWindow(int id ){
            this.id = id;
        }

        @Override
        public void run() {
            while(true){
                synchronized (lock){
                    if(remainingTickets > 0 ){  // 如果还有票，则可以购票
                        buyTicket();            // 购票
                    }else{
                        System.out.println("票已经售完，窗口" + this.id + "关闭。");
                        break;
                    }
                }
                // 模拟购票后的其他操作，增加窗口随机性
                try{
                    Thread.sleep((long) (Math.random()*1000));
                }catch (InterruptedException e){
                    e.printStackTrace();
                }
            }
        }
        
        // 购票方法
        private void buyTicket(){
            int buyTickets = Math.min((int) (Math.random() * 9 + 1), remainingTickets);
            remainingTickets -= buyTickets;
            System.out.println("从窗口 " + this.id + " 处购票 " + buyTickets + " 张， 剩余 " + remainingTickets +" 票。");
        }
    }

}
```
