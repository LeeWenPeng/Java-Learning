## 1  题目

- 两个线程交替打印
- 一个打印字母 abcd
- 一个打印数字 1234
- 目标 :a1b2c3…

## 2  代码

```java
package cn.msgcenter.test;

/**
 * @description: TODO
 * @author: Lee
 * @version: 1.0
 * @date: 2026-08-30 09:48
 */
public class AlternatePrinting3 {
    private static final Object lock = new Object();
    private static int state = 0; // 状态 0 letter 1 number
    private static int round = 0; // 当前打印轮数

    public static void main(String[] args) {
        // letter
        Thread thread = new Thread(() -> {
            for(int i = 0; i < 40; i++){
                synchronized (lock) {
                    while(state != 0){
                        try{
                            lock.wait();
                        }catch (InterruptedException e){
                            e.printStackTrace();
                        }
                    }
                    if(round < 10){
                        char letter = (char)('a' + (i%4));
                        System.out.print(letter);
                        state = 1;
                        lock.notifyAll();
                    }
                }
            }
        });

        Thread thread1 = new Thread(() -> {
            for(int i = 0; i < 40; i++){
                synchronized (lock) {
                    while(state != 1){
                        try{
                            lock.wait();
                        }catch (InterruptedException e){
                            e.printStackTrace();
                        }
                    }

                    if(round < 10){
                        int num = i%4 + 1;
                        System.out.print(num);
                        state = 0;

                        if(num == 4){
                            round++;
                        }

                        lock.notifyAll();
                    }
                }
            }
        });

        thread.start();
        thread1.start();
    }
}
```
