## 1  实验：Serializable

```java
package org.Serialize;

import java.io.*;
public class SerializeDemo1 {

    public static void main(String[] args) throws IOException, ClassNotFoundException {
        final String filename = "/home/leewenpeng/IdeaProjects/untitled/resources/text.dat";
        serialize(filename);
        Object obj = deserialize(filename);
        System.out.println(obj);
    }

	/**
     * 可序列化对象 继承 Serializable
     * 1. 默认的序列化机制
     */
    static class Person implements Serializable {

        enum Sex{
            MALE, FEMALE
        }

        private static final long serialVersionUID = 1L;

        private String name = null;
        private Integer age = null;
        private Sex sex;

        public Person(){}

        public Person(String name, Integer age, Sex sex) {
            this.name = name;
            this.age = age;
            this.sex = sex;
        }

        @Override
        public String toString() {
            return "Person{" +
                    "name='" + name + '\'' +
                    ", age=" + age +
                    ", sex=" + sex +
                    '}';
        }
    }

    /**
     * 序列化
     * @param filename 文件
     * @throws IOException
     */
    private static void serialize(String filename) throws IOException {
        File f = new File(filename);
        OutputStream out = new FileOutputStream(f);
        ObjectOutputStream oos = new ObjectOutputStream(out);
        oos.writeObject(new Person("Jack", 30, Person.Sex.MALE));
        oos.close();
        out.close();
    }

    /**
     * 反序列化
     * @param filename
     * @throws IOException
     */
    private static Object deserialize(String filename) throws IOException, ClassNotFoundException {
        File f = new File(filename);
        InputStream in = new FileInputStream(f);
        ObjectInputStream ois = new ObjectInputStream(in);
        Object obj = ois.readObject();
        ois.close();
        in.close();
        return obj;
    }

}
```

```java
	/**
	 - 可序列化对象，继承 Externalizable
	 - 要求
	 - 1. 提供一个 public 无参构造函数
	 - 2. 重写 writeExternal 和 readExternal 方法
	 */
    static class Person implements Externalizable {
        enum Sex{
            MALE, FEMALE
        }

        private static final long serialVersionUID = 1L;

        private String name = null;
        transient private Integer age = null; // transient
        private Person.Sex sex;

        public Person(){}

        public Person(String name, Integer age, Person.Sex sex) {
            this.name = name;
            this.age = age;
            this.sex = sex;
        }

        @Override
        public String toString() {
            return "Person{" +
                    "name='" + name + '\'' +
                    ", age=" + age +
                    ", sex=" + sex +
                    '}';
        }

        @Override
        public void writeExternal(ObjectOutput objectOutput) throws IOException {
            objectOutput.writeObject(name);
            objectOutput.writeInt(age);
            objectOutput.writeObject(sex);
        }

        @Override
        public void readExternal(ObjectInput objectInput) throws IOException, ClassNotFoundException {
            name = (String) objectInput.readObject();
            age = objectInput.readInt();

        }
    }

     * 序列化
     * @param filename 文件
     * @throws IOException
     */

        // 1. 设置保存路径
        File f = new File(filename);
        // 2.
        OutputStream out = new FileOutputStream(f);
        ObjectOutputStream oos = new ObjectOutputStream(out);
        oos.writeObject(new ExternlizaDemo1.Person("Jack", 30, Person.Sex.MALE));
        oos.close();
        out.close();
    }

    /**
     * 反序列化
     * @param filename
     * @throws IOException
     */

        File f = new File(filename);
        InputStream in = new FileInputStream(f);

        Object obj = ois.readObject();
```

## 3  实验：Serializable + writeObject + readObject

```java
package org.Serialize;

import java.io.*;

public class SerializeDemo1 {
    public static void main(String[] args) throws IOException, ClassNotFoundException {
        final String filename = "/home/leewenpeng/IdeaProjects/untitled/resources/text.dat";
        serialize(filename);
        Object obj = deserialize(filename);
        System.out.println(obj);
    }

    /**
     * 可序列化对象 继承 Serializable
     * 1. 重写了 writeObject 和 readObject
     * 2. 两个函数的方法签名必须严格遵循以下形式
     *      private void writeObject(ObjectOutputStream out)
     *      private void readObject(ObjectInputStream in)
     */
    static class Person implements Serializable {

        enum Sex{
            MALE, FEMALE
        }

        private static final long serialVersionUID = 1L;

        private String name = null;

        // transient
        transient private Integer age = null;
        private Sex sex;

        public Person(){}

        public Person(String name, Integer age, Sex sex) {
            this.name = name;
            this.age = age;
            this.sex = sex;
        }

        @Override
        public String toString() {
            return "Person{" +
                    "name='" + name + '\'' +
                    ", age=" + age +
                    ", sex=" + sex +
                    '}';
        }

        // Externalizable 的替代方案
        // 实现 Serializable 类 + 实现 writeObject\readObject 方法

        /**
         * 等价于 writeExternal 方法
         * 一般先调用 out.defaultWriteObject(); 以执行默认序列化
         * 然后再对 transient 关键字修饰的字段进行处理
         * @param out
         * @throws IOException
         */
        @Serial
        private void writeObject(ObjectOutputStream out) throws IOException {
            out.defaultWriteObject(); // 执行默认序列化
            out.writeInt(age);
        }

        @Serial
        private void readObject(ObjectInputStream in) throws IOException, ClassNotFoundException {
            in.defaultReadObject();
            age = in.readInt();
        }
    }

    /**
     * 序列化
     * @param filename 文件
     * @throws IOException
     */
    private static void serialize(String filename) throws IOException {
        // 1. 设置保存路径
        File f = new File(filename);
        // 2.
        OutputStream out = new FileOutputStream(f);
        ObjectOutputStream oos = new ObjectOutputStream(out);
        oos.writeObject(new Person("Jack", 30, Person.Sex.MALE));
        oos.close();
        out.close();
    }

    /**
     * 反序列化
     * @param filename
     * @throws IOException
     */
    private static Object deserialize(String filename) throws IOException, ClassNotFoundException {
        File f = new File(filename);
        InputStream in = new FileInputStream(f);
        ObjectInputStream ois = new ObjectInputStream(in);
        Object obj = ois.readObject();
        ois.close();
        in.close();
        return obj;
    }
}
```
