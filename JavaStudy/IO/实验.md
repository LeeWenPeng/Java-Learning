package org.IO;

import java.io.IOException;
import java.io.PrintStream;
import java.net.ServerSocket;
import java.net.Socket;
import java.util.Scanner;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

public class BIO{
    private static volatile boolean stop = false;
    private static final int MAX_CONNECTIONS =  2;

    public static void main(String[] args) throws IOException, InterruptedException {
        int connectionNum = 0;
        int port = 8888;


        ExecutorService service = Executors.newCachedThreadPool();
        try (ServerSocket serverSocket = new ServerSocket(port)){
            System.out.println("Server started on port " + port);

            while(!stop && connectionNum < MAX_CONNECTIONS){
                // 接受新的链接
                Socket socket = serverSocket.accept();
                connectionNum++;
                System.out.println("Accepted connection #" + connectionNum);

                if(connectionNum == MAX_CONNECTIONS){
                    stop = true;
                    System.out.println("Reached max connections, no further accepts.");
                }

                // 提交处理任务
                service.execute(
                        () -> handleClient(socket)
                );
            }
            System.out.println("Server stop accepting new connections. Waiting for active tasks to complete…");
        }finally {
            service.shutdown();
            if(!service.awaitTermination(5, TimeUnit.SECONDS)){
                service.shutdownNow();
            }
            System.out.println("Server shut down.");
        }
    }
    private static void handleClient(Socket socket){
        try(socket;
            Scanner scanner = new Scanner(socket.getInputStream());
            PrintStream printStream = new PrintStream(socket.getOutputStream());
        ) {
            // 读取客户端输入直到链接断开或服务器主动停止
            while(!stop && scanner.hasNextLine()){
                String line = scanner.nextLine().trim();
                if(line.isEmpty()){
                    continue;
                }
                printStream.println("PONG: " + line);
                if(printStream.checkError()){
                    break;
                }
            }

        } catch (IOException e) {
            System.out.println("Client handler error: " + e.getMessage());
            throw new RuntimeException(e);
        }
    }

package org.IO;

import java.io.IOException;
import java.net.InetSocketAddress;
import java.nio.ByteBuffer;
import java.nio.channels.SelectionKey;
import java.nio.channels.Selector;
import java.nio.channels.ServerSocketChannel;
import java.nio.channels.SocketChannel;
import java.util.Iterator;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;

/**
 * @BelongsProject: untitled
 * @BelongsPackage: org.IO
 * @Author: Lee
 * @CreateTime: 2026-04-23 23:07
 * @Description: TODO
 * @Version: 1.0
 */
public class NIO {
    private static volatile boolean stop = false;
    private static final int Max_CONNECT_NUM = 10;
    private static final int PORT = 8888;
    private static final String HOST = "127.0.0.1";
    private static final int BUFFER_SIZE = 1024;

    public static void main(String[] args) throws IOException {
        int connectionNum = 0;

        // 1. 打开并配置 ServerSocketChannel
        ServerSocketChannel ssc = ServerSocketChannel.open();
        ssc.configureBlocking(false); // 非阻塞模式
        ssc.socket().bind(new InetSocketAddress(HOST, PORT)); // 绑定 IP PORT

        // 2. 打开 事件监听器
        Selector selector = Selector.open();

        // 将服务器套接字通道 ssc 注册到多路复用器 selector 上，并且告诉 selector：我对这个通道本身支持的所有操作 都感兴趣。
//        ssc.register(selector, ssc.validOps()); // 等价于下面一行代码 ServerSocketChannel 只支持 SelectionKey.OP_ACCEPT（接受新连接）。
        // 3. 注册 ServerSocketChannel 到 Selector， 监听 OP_ACCEPT 事件
        ssc.register(selector, SelectionKey.OP_ACCEPT);

        /*

        四种事件类型
        1. OP_ACCEPT    新连接事件
        2. OP_CONNECT   连接就绪事件
        3. OP_READ      读事件
        4. OP_WRITE     写事件
            表示写就绪，当底层缓冲区有空闲，这个事件就会一直发生，浪费占用 CPU 资源
            一般不注册

        客户端和服务器可以是不同类型的模式 NIO 和 BIO
         */

        // 4. 创建线程池
        ExecutorService pool = Executors.newCachedThreadPool();

        while (!stop) {
            // select 阻塞等待事件
            // 一旦有新的事件到达，程序才会继续向下执行
            int num = selector.select();
            if(num == 0){
                continue;
            }

            // 思考题：如果事件不删除的话，或者漏掉了某个事件的处理，会有什么后果？

            // 获取就绪的 selectedKeys 集合
            Iterator<SelectionKey> events = selector.selectedKeys().iterator();
            while(events.hasNext()){
                SelectionKey event = events.next();

                if(event.isAcceptable()) {
                    // 连接事件，获取新连接
                    ServerSocketChannel server = (ServerSocketChannel) event.channel();
                    SocketChannel client = server.accept();

                    if(client != null) {
                        if(connectionNum >= Max_CONNECT_NUM){
                            // 连接数超过最大
                            client.close();
                            System.out.println("已经达到最大的连接数");
                        }else{

                            // 为新连接设置为非阻塞
                            client.configureBlocking(false);
                            // 为新连接注册 READ 事件，并绑定一个空 ByteBuffer
                            client.register(selector, SelectionKey. , ByteBuffer.allocate(BUFFER_SIZE));
                            connectionNum++;
                            System.out.println("接受新的连接 #" + connectionNum + " : " + client.getRemoteAddress());

                            if(connectionNum == Max_CONNECT_NUM){
                                // 连接数达到上限，设置停止标志
                                stop = true;
                                System.out.println("已经达到最大的连接数, 不再接受新的连接");
                            }
                        }
                    }

                }else if(event.isReadable()) {
                    // 可读事件

                    // 获取可读的 socket
                    SocketChannel client = (SocketChannel) event.channel();
//                        ByteBuffer buffer = ByteBuffer.allocate(1024);
                    // 设置读缓冲区，由于之间加入连接时已经为新可读连接添加了附件，这里取出附件作为读缓冲区
                    ByteBuffer buffer = (ByteBuffer) event.attachment();

                    try {
                        int byteRead = client.read(buffer);
                        if (byteRead == -1
) {
                            // 没有读到字节，将客户端关闭
                            closeClient(client, event);
                            connectionNum--;
                            continue;
                        } else if (byteRead > 0) {
                            // 准备读取数据，翻转 buffer
                            buffer.flip();
                            byte[] data = new byte[buffer.remaining()];
                            buffer.get(data);
                            String received = new String(data).trim();
                            System.out.println("收到：" + received);

                            // 构造响应
                            String response = "PONG: " + received + "\n";
                            ByteBuffer writeBuffer = ByteBuffer.wrap(response.getBytes());
                            event.attach(writeBuffer); // 将要写的数据挂到 event 的附件中
                            event.interestOps(SelectionKey.OP_WRITE | SelectionKey.OP_READ); // 注册写事件
//                            client.write(writeBuffer); // 尝试直接写出
//                            if (writeBuffer.hasRemaining()) {
//                                // 如果没写完，则将没写完的数据挂到 event 的附件中，并向 selector 增加写事件
//                                event.attach(writeBuffer);
//                                event.interestOps(SelectionKey.OP_WRITE | SelectionKey.OP_READ);
//                            } else {
//                                // 写完了，清空 buffer，继续只关注读事件
//                                buffer.clear();
//                                event.attach(buffer);
//                                event.interestOps(SelectionKey.OP_READ);
//                            }
                        }
                    }catch (IOException e) {
                        // 客户端异常断开
                        closeClient(client, event);
                        connectionNum--;
                    }
                }else if(event.isWritable()) {
                    // 可写事件

                    SocketChannel client = (SocketChannel) event.channel();
                    ByteBuffer writeBuffer = (ByteBuffer) event.attachment();

                    try{
                        client.write(writeBuffer);
                        if(!writeBuffer.hasRemaining()){
                            // 全部写完，切换为 READ 模式
                            ByteBuffer readBuffer = ByteBuffer.allocate(BUFFER_SIZE);
                            event.attach(readBuffer);
                            event.interestOps(SelectionKey.OP_READ);
                        }
                    }catch (IOException e){
                        closeClient(client, event);
                        connectionNum--;
                    }
                }
                // 移除已经解决的事件
                events.remove();
            }
        }
        // 优雅关闭
        System.out.println("服务器正在关闭 …");
        pool.shutdown();
        selector.close();
        ssc.close();
        System.out.println("服务器已经停止!");
    }
    // 安全关闭客户端连接
    private static void closeClient(SocketChannel client, SelectionKey event) throws IOException {
        System.out.println("关闭连接" + client.getRemoteAddress());
        event.cancel(); // 取消注册
        client.close(); // 关闭通道
    }
}

package org.IO;

import java.io.IOException;
import java.net.InetSocketAddress;
import java.nio.Buffer;
import java.nio.ByteBuffer;
import java.nio.channels.AsynchronousServerSocketChannel;
import java.nio.channels.AsynchronousSocketChannel;
import java.nio.channels.CompletionHandler;

/**
 * @BelongsProject: untitled
 * @BelongsPackage: org.IO
 * @Author: Lee
 * @CreateTime: 2026-04-26 18:52
 * @Description: TODO
 * @Version: 1.0
 */
public class AIO {
    public static void main(String[] args) throws IOException, InterruptedException {
        int port = 8888;
        String address = "127.0.0.1";

        AsynchronousServerSocketChannel serverSocketChannel = AsynchronousServerSocketChannel.open();
        serverSocketChannel.bind(new InetSocketAddress(address, port));

        System.out.println("AIO 服务器启动，监听 " + address + ":" + port);

        // 开始接受新连接，使用自定义的 CompletionHandler 实现类
        serverSocketChannel.accept(null, new AcceptHandler(serverSocketChannel));

        // 保持主线程不退出
        Thread.sleep(Integer.MAX_VALUE);
    }
    static class AcceptHandler implements CompletionHandler<AsynchronousSocketChannel, Object> {
        private final  AsynchronousServerSocketChannel serverSocketChannel;

        public AcceptHandler(AsynchronousServerSocketChannel serverSocketChannel) {
            this.serverSocketChannel = serverSocketChannel;
        }

        @Override
        public void completed(AsynchronousSocketChannel clientChannel, Object attachment) {
            // 继续接受下一个新连接
            serverSocketChannel.accept(attachment, this);

            System.out.println("客户端连接: " + clientChannel);

            // 开始读写处理
            handleClient(clientChannel);
        }

        @Override
        public void failed(Throwable exc, Object object) {
            exc.printStackTrace();
        }

        /**
         * 读写处理
         * @param clientChannel
         */
        private void handleClient(AsynchronousSocketChannel clientChannel) {
            // 每个连接维持一个独立的缓冲区
            ByteBuffer readBuffer = ByteBuffer.allocate(1024);
            // 启动异步读，使用自定义的 CompletionHandler 类
            clientChannel.read(readBuffer, readBuffer, new ReadHandler(clientChannel, readBuffer));
        }
    }

    static class ReadHandler implements CompletionHandler<Integer, Object> {
        private final AsynchronousSocketChannel clientChannel;
        private final ByteBuffer readBuffer;

        public ReadHandler(AsynchronousSocketChannel clientChannel, ByteBuffer readBuffer) {
            this.clientChannel = clientChannel;
            this.readBuffer = readBuffer;
        }

        @Override
        public void completed(Integer bytesRead, Object attachment) {
            if(bytesRead == -1) {
                // 客户端关闭
                closeChannel(clientChannel);
                return;
            }

            // 切换为读模式，从缓冲区读取数据
            readBuffer.flip();
            byte[] bytes = new byte[readBuffer.remaining()];
            readBuffer.get(bytes);
            String received = new String(bytes);
            System.out.println("收到: " + received);

            // 响应数据
            String responseMes = "PONG: " + received;
            ByteBuffer responseBuffer = ByteBuffer.wrap(responseMes.getBytes());

            // 异步写
            clientChannel.write(responseBuffer, null, new CompletionHandler<Integer, Object>() {
                @Override
                public void completed(Integer integer, Object object) {
                    // 写完成，清理读缓冲区，准备下一次读取
                    readBuffer.clear();
                    clientChannel.read(readBuffer, null, ReadHandler.this);
                }

                @Override
                public void failed(Throwable throwable, Object object) {
                    // 写失败
                    throwable.printStackTrace();
                    closeChannel(clientChannel);
                }
            });
        }

        @Override
        public void failed(Throwable throwable, Object object) {
            throwable.printStackTrace();
            closeChannel(clientChannel);
        }

        private void closeChannel(AsynchronousSocketChannel channel){
            try{
                System.out.println("closeChannel: " + channel);
                channel.close();
            }catch (IOException e){
                e.printStackTrace();
            }
        }
    }
}
