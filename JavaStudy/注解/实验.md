```java
package org.Annotations;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

/*
    ElementType.TYPE 类
    METHOD 方法
    CONSTRUCTOR 构造器
    FIELD 成员变量

    RetentionPolicy.SOURCE 该注解只在编译前保留，进行编译时会被删除。也就是说，该注解就是注释
    RetentionPolicy.CLASS 该注解在编译时生效，一旦程序运行，该注解就会被删除
    RetentionPolicy.RUNTIME 该注解在运行时生效,可以配合反射

 */

//@Target(ElementType.TYPE) // 限制注解能够注解的范围，或者说使用目标
@Target({ElementType.TYPE, ElementType.METHOD})
@Retention(RetentionPolicy.RUNTIME) // 运行时保持
public @interface MyOwnAnnotation {

}
```

### 1.3  注解 RunImmediately

```java
package org.Annotations;

import java.lang.annotation.ElementType;
import java.lang.annotation.Retention;
import java.lang.annotation.RetentionPolicy;
import java.lang.annotation.Target;

@Target(ElementType.METHOD)
@Retention(RetentionPolicy.RUNTIME)
public @interface RunImmediately {
    // 参数类型只能为 基本数据类型
    int times() default 1;
}
```

```java
/*
        根据注解执行函数
     */
    <A extends Annotation>void RunImmediatelyTimesMethod(Object obj, Class<A> annoClass) throws NoSuchMethodException, InvocationTargetException, InstantiationException, IllegalAccessException {
        Method[] methods = obj.getClass().getDeclaredMethods();
        for(Method method : methods){
            A anno = method.getAnnotation(annoClass);
            if(method.isAnnotationPresent(annoClass)){
                Method timesMethod = annoClass.getMethod("times");
                int fin = (int) timesMethod.invoke(anno);
                for(int i = 0; i < fin; ++i){
                    method.invoke(obj);
                }
            }
        }
    }
```

```java
        根据注解获取字段

     */
    <A extends Annotation>String getImportantString(Object obj, Class<A> annoClass) throws IllegalAccessException {
        String res = "";

        Field[] fields = obj.getClass().getDeclaredFields();
        for(Field field : fields){
            if(field.isAnnotationPresent(annoClass)){
                field.setAccessible(true);
                Object value = field.get(obj);
                if(value instanceof String strValue){
                    res = strValue;
                }
            }
        }
        return res.toUpperCase();
    }
```

### 2.3  主函数

```java
package org.DAO;

import java.io.ObjectStreamException;
import java.lang.reflect.Field;
import java.lang.reflect.InvocationTargetException;
import java.lang.reflect.Method;

/**
 * @BelongsProject: untitled
 * @BelongsPackage: org.DAO
 * @Author: Lee
 * @CreateTime: 2026-04-19 16:13
 * @Description: TODO
 * @Version: 1.0
 */
public class Test {
    public static void main(String[] args) throws InvocationTargetException, IllegalAccessException, NoSuchMethodException {
        Filter f1 = new Filter();
        f1.setId(10);

        Filter f2 = new Filter();
        f2.setUserName("lucy");

        Filter f3 = new Filter();
//        f3.setEmail("xxx@gmail.com, xxx2@gmail.com, xxx3@gmail.com");

        f3.setEmail("xxx@gmail.com");
        f3.setAge(12);

        String sq1 = query(f1);
        String sq2 = query(f2);
        String sq3 = query(f3);

        System.out.println(sq1);
        System.out.println(sq2);
        System.out.println(sq3);

        Filter2 f21 = new Filter2();
        f21.setAmount(1000);
        System.out.println(query(f21));
    }

    /**
     * 返回 f 的 SQL 查询语句
     * 1. 考虑代码怎么和数据库做映射
     *  使用注解
     *      Table(表名) 标注在 Bean 类上
     *      Column(段名) 标注在 Bean 字段上
     * @param f 要查询的对象的条件
     * @return 根据 f 查询到对象的SQL 语句
     */
    private static String query(Object f) throws IllegalAccessException, NoSuchMethodException, InvocationTargetException {
        StringBuilder sb = new StringBuilder();

        Class<?> clazz = f.getClass();
        if(!clazz.isAnnotationPresent(Table.class)){
            return null;
        }
        Table table = clazz.getAnnotation(Table.class);
        String tableName = table.value();

        sb.append("select * from").append(tableName).append(" where 1 = 1"); // 添加 where 1 = 1 防止后续没有字段报错

        // 3. 遍历所有字段，如果有值，就添加
        Field[] fields = clazz.getDeclaredFields();
        for(Field field : fields){
            // 4. 处理每个字段
            // 4.1 获取字段名
            if(!field.isAnnotationPresent(Column.class)){
                System.err.println("非字段");
                continue;
            }
            Column column = field.getAnnotation(Column.class);
            String columnName = column.value(); // 字段名
            // 4.2 获取值：通过反射调用
            String fieldName = field.getName();
            String getMethodName = "get" + fieldName.substring(0, 1).toUpperCase() + fieldName.substring(1);
            Method getFieldValueMethod = clazz.getMethod(getMethodName);
            Object val =  getFieldValueMethod.invoke(f);
            // 4.3 拼装 SQL
            if(val == null || val instanceof Integer && (int) val == 0){
                continue;
            }

            if(fieldName.equals("email")){
                String[] subs = ((String) val).split(",\\s*");
                sb.append(" and ").append(columnName).append(" in (");
                for(String sub : subs){
                    sb.append("'").append(sub).append("', ") ;
                }
                sb.delete(sb.length()-2, sb.length());
                sb.append(")");
                continue;
            }
            sb.append(" and ").append(columnName).append("=").append(val);
        }
        return sb.toString();
    }
}
```

### 3.3  util

```java
package org.Annotation.RegexValidAnno;

import java.lang.reflect.Field;
import java.util.regex.Matcher;
import java.util.regex.Pattern;

/**
 * @BelongsProject: untitled
 * @BelongsPackage: org.Annotation.RegexValidAnno
 * @Author: Lee
 * @CreateTime: 2026-04-20 16:23
 * @Description: 注解处理器
 * @Version: 1.0
 */
public class RegexValidUtil {
    public static boolean check(Object obj) throws Exception {

        boolean res = true; // 是否合法
        StringBuilder sb = new StringBuilder(); // 记录报错信息

        Field[] fields = obj.getClass().getDeclaredFields(); // 字段列表
        for(Field field : fields){
            // 1. 判断 field 是否被 @RegexValid 修饰
            if(!field.isAnnotationPresent(RegexValid.class)){
                // 没有被标注，该字段不用解析
                continue;
            }

            // 2. 获取注解
            RegexValid regexValid = field.getAnnotation(RegexValid.class);

            // 3. 获取 value，对照组
            String value = regexValid.value();
            if(value.isEmpty()){ // value 是空,则 value = policy.policy
                Policy policy = regexValid.policy();
                value = policy.getPolicy();
            }

            // 4. 获取 field 的值，实验组
            field.setAccessible(true);
            Object fieldValue = field.get(obj);
            if(fieldValue == null) {
                // 目标值为空，空值错误
                sb.append("\n").append(obj.getClass().getName()).append(" 类中的 ").append(field.getName()).append(" 不能为空！！！");
                res = false;
                break;
            }

            // 5. 判断是否为合法正则表达式
            if(fieldValue instanceof String text){
                Pattern p = Pattern.compile(value);
                Matcher m = p.matcher(text);
                res = m.matches(); // 是否是合法的正则表达式
                if(!res){
                    // 非法正则
                    sb.append("\n").append(obj.getClass().getName()).append(" 类中的 ").append(field.getName()).append(" 不是合法字符串！！！");
                    break;
                }
            }else{
                // 获取到的值类型非字符串
                sb.append("\n").append(obj.getClass().getName()).append(" 类中的 ").append(field.getName()).append(" 不是字符串类型，不能使用此注解校验！！！");
                res = false;
                break;
            }
        }
        if(!sb.isEmpty()){
            throw new Exception(sb.toString());
        }

        return res;
    }
}
```
